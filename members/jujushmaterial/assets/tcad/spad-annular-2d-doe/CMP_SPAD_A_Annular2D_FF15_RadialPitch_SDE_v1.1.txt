; PURPOSE

;   radial pitches for A's proposed concentric annular Si/STI optical pattern.
;
; RESEARCH CONSTRAINTS (FIXED)
;   - Frozen common baseline electrical core is NOT recalibrated.
;   - Native STI1/STI2/STI3/STI4/STI5 positions and widths remain unchanged.
;   - STI depth remains fixed at 0.300 um.
;   - Reference SPAD itself has NO optical pattern FF; Reference FF is N/A.
;   - The intentional ANNULAR optical pattern has Si-area FF = 15%.
;   - Workbench independent variable: @RadialPitch@ [um].
;   - Annular Si ring width is a DEPENDENT value solved automatically so that
;     the ideal concentric annular pattern area fraction is exactly 0.15.
;   - Pattern phase is fixed: each radial cell starts with Silicon, followed by
;     an intentional STI interval. Phase is NOT swept.
;
; 2D INTERPRETATION
;   This deck is a diametric cut through the intended concentric annular pattern.
;   Sentaurus 2D EMW extrudes this cut in the third direction; therefore this is
;   a screening model, NOT an exact 3D annular optical result. Final optical
;   claims require 3D annular SDE/EMW on selected candidates.
;
; FF DEFINITION
;   [surrogate design definition]
;   The intentional annular-pattern aperture is r = 0.55 ... 12.20 um, matching
;   the baseline SOI optical-stack span from the anode-window edge to STI5.
;   The PRIMARY FF constraint is the intended 3D annular optical-pattern Si area
;   divided by this pattern aperture area. It is solved to exactly 0.15.
;   Native baseline STI1-5 are NOT compensated away by widening the new rings.
;   Their overlap with intended Si rings is reported separately as a diagnostic
;   FinalSiFraction_AfterNativeSTI. Reference FF remains N/A.
;
; All lengths are in um. Doping concentrations are in cm^-3.

; -----------------------------------------------------------------------------
; A proposed annular optical-pattern design variables
; -----------------------------------------------------------------------------
; IMPORTANT:
;   Reference FF = N/A.
;   FF_TARGET below belongs ONLY to the intentional proposed annular pattern.
(define FF_TARGET 0.15)
(define RadialPitch @RadialPitch@)

; Pattern aperture follows the baseline SOI optical-stack span.
; [surrogate design definition]
(define r_pattern_inner x_anode_end)
(define r_pattern_outer x_STI5)
(define pattern_span (- r_pattern_outer r_pattern_inner))

; The aperture can terminate part-way through the last pitch.  The exact ring
; width is therefore solved from 3D ANNULUS AREA, not approximated as FF*pitch.
; Pi cancels in all area ratios, so area_no_pi = (r_outer^2-r_inner^2).
(define N_full (floor (/ pattern_span RadialPitch)))
(define residual (- pattern_span (* N_full RadialPitch)))
(define pattern_area_no_pi
  (- (* r_pattern_outer r_pattern_outer)
     (* r_pattern_inner r_pattern_inner)))
(define target_si_area_no_pi (* FF_TARGET pattern_area_no_pi))

; B = sum of inner radii of the N complete radial cells.
(define B_full
  (+ (* N_full r_pattern_inner)
     (* 0.5 RadialPitch N_full (- N_full 1.0))))
(define r_last (+ r_pattern_inner (* N_full RadialPitch)))

; Candidate solution when the final partial cell contains only part of its Si ring
; (RingWidth <= residual).
(define C_partial (+ B_full r_last))
(define RingWidth_case1
  (/ (+ (- C_partial)
        (sqrt (+ (* C_partial C_partial)
                 (* (+ N_full 1.0) target_si_area_no_pi))))
     (+ N_full 1.0)))

; Candidate solution when the final residual is entirely Si (RingWidth > residual).
(define residual_si_area_no_pi
  (+ (* 2.0 r_last residual) (* residual residual)))
(define RingWidth_case2
  (/ (+ (- B_full)
        (sqrt (+ (* B_full B_full)
                 (* N_full
                    (- target_si_area_no_pi residual_si_area_no_pi)))))
     N_full))

(define RingWidth
  (if (and (> residual 0.0) (<= RingWidth_case1 residual))
      RingWidth_case1
      RingWidth_case2))

(define AnnularSTIWidth (- RadialPitch RingWidth))

; Independent exact 3D-area check of the intentional optical-pattern FF.
(define full_si_area_no_pi
  (+ (* N_full RingWidth RingWidth)
     (* 2.0 RingWidth B_full)))
(define partial_si_width
  (if (> residual 0.0)
      (if (< residual RingWidth) residual RingWidth)
      0.0))
(define partial_si_area_no_pi
  (+ (* 2.0 r_last partial_si_width)
     (* partial_si_width partial_si_width)))
(define ideal_pattern_si_area_no_pi
  (+ full_si_area_no_pi partial_si_area_no_pi))
(define PatternFF_3D
  (/ ideal_pattern_si_area_no_pi pattern_area_no_pi))
(define PatternFF_error (- PatternFF_3D FF_TARGET))

; -----------------------------------------------------------------------------
; Native-STI overlap diagnostic ONLY.
; -----------------------------------------------------------------------------
; This does NOT alter RingWidth.  It quantifies how much of the intended 15% Si
; annular-pattern area is already occupied by frozen native STI in the final
; material map.  Thus the primary comparison remains FF_pattern=15% for every
; radial pitch, while baseline-topology interaction is visible as a diagnostic.

(define (rmin a b) (if (< a b) a b))
(define (rmax a b) (if (> a b) a b))

; Radial annulus intersection area without pi.
(define (annulus-overlap-no-pi a0 a1 b0 b1)
  (let* ((lo (rmax a0 b0))
         (hi (rmin a1 b1)))
    (if (> hi lo)
        (- (* hi hi) (* lo lo))
        0.0)))

; Frozen native STI radial intervals on the right half.  STI5 begins exactly at
; r_pattern_outer, so it should contribute zero but is included as a guard/check.
(define (native-overlap-for-si-ring r0 r1)
  (+
    (annulus-overlap-no-pi r0 r1 x_STI1 (+ x_STI1 w_STI1))
    (annulus-overlap-no-pi r0 r1 x_STI2 (+ x_STI2 w_STI_active))
    (annulus-overlap-no-pi r0 r1 x_STI3 (+ x_STI3 w_STI_active))
    (annulus-overlap-no-pi r0 r1 x_STI4 (+ x_STI4 w_STI_active))
    (annulus-overlap-no-pi r0 r1 x_STI5 (+ x_STI5 w_STI5))))

(define (sum-native-overlap-full-cells k)
  (if (< k N_full)
      (let* ((r0 (+ r_pattern_inner (* k RadialPitch)))
             (r1 (+ r0 RingWidth)))
        (+ (native-overlap-for-si-ring r0 r1)
           (sum-native-overlap-full-cells (+ k 1))))
      0.0))

(define native_overlap_full_no_pi
  (sum-native-overlap-full-cells 0))
(define native_overlap_partial_no_pi
  (if (> partial_si_width 0.0)
      (native-overlap-for-si-ring
        r_last (+ r_last partial_si_width))
      0.0))
(define NativeOverlapArea_no_pi
  (+ native_overlap_full_no_pi native_overlap_partial_no_pi))

(define FinalSiAreaAfterNative_no_pi
  (- ideal_pattern_si_area_no_pi NativeOverlapArea_no_pi))
(define FinalSiFraction_AfterNativeSTI
  (/ FinalSiAreaAfterNative_no_pi pattern_area_no_pi))
(define NativeOverlapFraction_Aperture
  (/ NativeOverlapArea_no_pi pattern_area_no_pi))
(define NativeOverlapFraction_PatternSi
  (if (> ideal_pattern_si_area_no_pi 0.0)
      (/ NativeOverlapArea_no_pi ideal_pattern_si_area_no_pi)
      0.0))

; -----------------------------------------------------------------------------
; 7.1 Add intentional annular-pattern STI on the RIGHT radial half
; -----------------------------------------------------------------------------
; Each radial period starts with a Silicon ring of dependent width RingWidth.
; The remainder of the period is converted to Oxide with the same vertical STI
; extent as the fixed process STI.  Existing native STI bodies are left in place.
;
; In this 2D deck these rectangles are the diametric cross-section of the
; intended 3D concentric annular STI trenches.  They are mirrored through x=0
; together with the frozen baseline bodies below.

(define (annular-sti-name k)
  (string-append "R.AnnularOpticalSTI." (number->string k)))

(define (build-annular-full-cells k)
  (if (< k N_full)
      (let* ((cell_r0 (+ r_pattern_inner (* k RadialPitch)))
             (sti_r0 (+ cell_r0 RingWidth))
             (sti_r1 (+ cell_r0 RadialPitch)))
        (if (> sti_r1 sti_r0)
            (make-rect (annular-sti-name k) "Oxide"
              sti_r0 y_uSi_top sti_r1 y_interface))
        (build-annular-full-cells (+ k 1)))))

(build-annular-full-cells 0)

; If the aperture ends inside an STI portion of the last partial period,
; create only the physically remaining fragment.  If residual <= RingWidth,
; the aperture ends inside Silicon and no terminal STI fragment is required.
(if (> residual RingWidth)
    (make-rect "R.AnnularOpticalSTI.Edge" "Oxide"
      (+ r_last RingWidth) y_uSi_top r_pattern_outer y_interface))
