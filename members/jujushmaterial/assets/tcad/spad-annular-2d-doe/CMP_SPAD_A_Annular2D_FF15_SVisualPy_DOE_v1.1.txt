# PURPOSE
#   For every Workbench RadialPitch split, extract 940 nm OpticalGeneration and
#   publish compact DOE metrics for A's 2D annular radial-pitch screening.
#
# DESIGN RULE
#   - Reference FF is N/A.
#   - Intentional ANNULAR optical-pattern FF is fixed at 0.15.
#   - RadialPitch is the ONLY A geometry sweep variable here.
#   - RingWidth is recalculated from exact finite-aperture annulus area.
#   - Frozen native STI is NOT compensated by widening RingWidth.
#   - Native overlap is reported separately as a geometry diagnostic.
#
# PRIMARY RANKING
#   G_SCR_Int / Gain_Ref are primary. Hotspot values/coordinates are diagnostics.
#   G180_Avg and G180_BandAvg monitor alignment around the frozen ~180.203 nm SCR
#   center but a single local maximum is not used alone to select the optimum.
#
# 2D LIMITATION
#   This is a diametric/radial cross-section screening result, not exact 3D
#   annular optics. Selected pitch regions must be taken to 3D EMW.

# [surrogate design definition] Same patternable aperture used by the SDE deck.
R_PATTERN_INNER_UM = 0.55
R_PATTERN_OUTER_UM = 12.20

# Frozen native STI radial intervals on the RIGHT half [um].
# These are geometry diagnostics only; they do not change the 15% pattern FF.
NATIVE_STI_INTERVALS_UM = (
    (0.700, 1.300),   # STI1
    (2.975, 3.275),   # STI2
    (6.100, 6.400),   # STI3
    (9.225, 9.525),   # STI4
    (12.200, 12.800), # STI5; starts at pattern outer edge -> nominal overlap 0
)


def _annulus_interval_overlap_no_pi(a0, a1, b0, b1):
    """Area/pi of intersection between two radial annulus intervals."""
    lo = max(float(a0), float(b0))
    hi = min(float(a1), float(b1))
    if hi <= lo:
        return 0.0
    return hi*hi - lo*lo


def _native_overlap_no_pi(r0, r1):
    return float(sum(
        _annulus_interval_overlap_no_pi(r0, r1, s0, s1)
        for s0, s1 in NATIVE_STI_INTERVALS_UM
    ))


def annular_ring_width_for_global_ff(pitch_um, ff=FF_TARGET,
                                     r_in=R_PATTERN_INNER_UM,
                                     r_out=R_PATTERN_OUTER_UM):
    """Solve ring width for exact 3D annular pattern FF in a finite aperture.

    Primary constraint:
        intended annular Si area / pattern aperture area = ff

    Native baseline STI overlap is computed AFTER the 15% pattern geometry is
    solved and is diagnostic only. Ring width is never widened to compensate.
    Pi cancels from the area-ratio equations.
    """
    T = float(pitch_um)
    if not (T > 0.0):
        raise RuntimeError(f"RadialPitch must be >0, got {T}")

    span = float(r_out - r_in)
    n = int(np.floor(span / T))
    if n < 1:
        raise RuntimeError(
            f"RadialPitch={T} um is too large for pattern span={span} um"
        )

    residual = float(span - n*T)
    area_total_no_pi = float(r_out*r_out - r_in*r_in)
    target_no_pi = float(ff * area_total_no_pi)
    B = float(n*r_in + 0.5*T*n*(n-1))
    r_last = float(r_in + n*T)

    if residual > 0.0:
        C = B + r_last
        w1 = float((-C + np.sqrt(C*C + (n+1)*target_no_pi)) / (n+1))
    else:
        w1 = np.nan

    if residual > 0.0 and w1 <= residual:
        w = w1
    else:
        partial_full_si_no_pi = 2.0*r_last*residual + residual*residual
        radicand = B*B + n*(target_no_pi-partial_full_si_no_pi)
        if radicand < -1.0e-14:
            raise RuntimeError(
                f"Negative ring-width radicand: {radicand} for pitch={T} um"
            )
        radicand = max(radicand, 0.0)
        w = float((-B + np.sqrt(radicand)) / n)

    if not (0.0 < w <= T):
        raise RuntimeError(
            f"Unphysical RingWidth={w} um for RadialPitch={T} um"
        )

    partial_si_width = min(max(residual, 0.0), w)
    full_si_no_pi = float(n*w*w + 2.0*w*B)
    partial_si_no_pi = float(
        2.0*r_last*partial_si_width + partial_si_width**2
    )
    ideal_si_no_pi = full_si_no_pi + partial_si_no_pi
    pattern_ff_3d = float(ideal_si_no_pi / area_total_no_pi)

    # Sum the intended-Si/native-STI annulus intersections. Native intervals are
    # disjoint in the frozen surrogate geometry, so no double-count correction
    # is needed.
    native_overlap_no_pi = 0.0
    for k in range(n):
        ring_r0 = float(r_in + k*T)
        ring_r1 = float(ring_r0 + w)
        native_overlap_no_pi += _native_overlap_no_pi(ring_r0, ring_r1)

    if partial_si_width > 0.0:
        native_overlap_no_pi += _native_overlap_no_pi(
            r_last, r_last + partial_si_width
        )

    final_si_no_pi = ideal_si_no_pi - native_overlap_no_pi
    final_si_fraction = float(final_si_no_pi / area_total_no_pi)
    native_overlap_fraction_aperture = float(
        native_overlap_no_pi / area_total_no_pi
    )
    native_overlap_fraction_pattern_si = float(
        native_overlap_no_pi / ideal_si_no_pi
    ) if ideal_si_no_pi > 0.0 else 0.0

    return {
        "pitch_um": T,
        "ring_width_um": w,
        "sti_width_um": T-w,
        "n_full": n,
        "residual_um": residual,
        "pattern_ff_3d": pattern_ff_3d,
        "pattern_ff_error": pattern_ff_3d - float(ff),
        "edge_sti_fragment_um": max(0.0, residual-w),
        "native_overlap_area_no_pi": float(native_overlap_no_pi),
        "native_overlap_area_um2": float(np.pi*native_overlap_no_pi),
        "native_overlap_fraction_aperture": native_overlap_fraction_aperture,
        "native_overlap_fraction_pattern_si": native_overlap_fraction_pattern_si,
        "final_si_fraction_after_native_sti": final_si_fraction,
    }

def is_svisual_error(obj):
    if not isinstance(obj, str):
        return False
    s = obj.strip().lower()
    return (
        s.startswith("error:") or
        s.startswith("validation failed") or
        "reference resolving failed" in s or
        "does not exist" in s
    )


def normalize_dataset_id(obj):
    if isinstance(obj, (list, tuple)):
        if len(obj) == 1:
            return obj[0]
        return obj
    return obj


def safe_list_variables(dataset):
    ds = normalize_dataset_id(dataset)
    result = sv.list_variables(dataset=ds)
    if is_svisual_error(result):
        raise RuntimeError(f"{ds}: list_variables returned error: {result}")
    if isinstance(result, str):
        return [result]
    return list(result)


def pickvar(dataset, patterns):
    names = safe_list_variables(dataset)
    for pat in patterns:
        p = pat.lower()
        for name in names:
            if fnmatch.fnmatch(name.lower(), p):
                return name
    return None


def get_data(dataset, varname):
    ds = normalize_dataset_id(dataset)
    return np.asarray(
        sv.get_variable_data(varname=varname, dataset=ds),
        dtype=float
    )


def finite_xy(x, y):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)
    m = np.isfinite(x) & np.isfinite(y)
    return x[m], y[m]


def sort_unique_average(x, y):
    x, y = finite_xy(x, y)
    idx = np.argsort(x, kind="mergesort")
    x = x[idx]
    y = y[idx]
    xu, inv = np.unique(x, return_inverse=True)
    sums = np.bincount(inv, weights=y)
    counts = np.bincount(inv)
    yu = sums / np.maximum(counts, 1)
    return xu, yu


def metric_stats(values):
    v = np.asarray(values, dtype=float)
    v = v[np.isfinite(v)]
    if v.size == 0:
        raise RuntimeError("No finite values for statistics")
    mean = float(np.mean(v))
    std = float(np.std(v))
    return {
        "count": int(v.size),
        "mean": mean,
        "min": float(np.min(v)),
        "max": float(np.max(v)),
        "std": std,
        "cv": float(std / mean) if mean != 0.0 else np.nan,
    }


def save_csv(filename, columns):
    names = list(columns.keys())
    arrays = [np.asarray(columns[name], dtype=float) for name in names]
    if not arrays:
        return
    n = len(arrays[0])
    if any(len(a) != n for a in arrays):
        raise RuntimeError(
            f"{filename}: column-length mismatch {[len(a) for a in arrays]}"
        )
    np.savetxt(
        filename,
        np.column_stack(arrays),
        delimiter=",",
        header=",".join(names),
        comments="",
        fmt="%.12e"
    )


def write_metric(fh, name, value):
    try:
        fv = float(value)
    except Exception:
        fh.write(f"{name} = {value}\n")
        return
    if np.isfinite(fv):
        fh.write(f"{name} = {fv:.12g}\n")
    else:
        fh.write(f"{name} = NA\n")


def emit(name, value):
    """Publish a compact Workbench DOE variable."""
    try:
        fv = float(value)
    except Exception:
        print(f"DOE: {name} x")
        return
    if np.isfinite(fv):
        print(f"DOE: {name} {fv:.12g}")
    else:
        print(f"DOE: {name} x")


def create_horizontal_cut(source_plot, source_ds, at_y, name):
    attempts = [
        ("plot+Silicon", dict(
            at=at_y, plot=source_plot, materials=["Silicon"], name=name)),
        ("plot-only", dict(
            at=at_y, plot=source_plot, name=name)),
        ("dataset+Silicon", dict(
            at=at_y, dataset=source_ds, materials=["Silicon"], name=name)),
        ("dataset-only", dict(
            at=at_y, dataset=source_ds, name=name)),
    ]

    errors = []
    for label, kwargs in attempts:
        try:
            result = sv.create_cutline("y", **kwargs)
        except Exception as exc:
            errors.append(f"{label}: exception={exc}")
            continue

        if is_svisual_error(result):
            errors.append(f"{label}: returned={result}")
            continue

        ds = normalize_dataset_id(result)
        print(
            f"EMW_ANNULAR_DOE: CUT y={at_y:.12g} "
            f"method={label} dataset={ds}"
        )
        return ds

    raise RuntimeError(
        f"Horizontal cutline failed at y={at_y}: " + " || ".join(errors)
    )


def extract_active_cutline(source_plot, source_ds, at_y, cut_name):
    cut_ds = create_horizontal_cut(
        source_plot=source_plot,
        source_ds=source_ds,
        at_y=at_y,
        name=cut_name
    )

    xname = pickvar(
        cut_ds,
        [
            "X", "x", "X-Coordinate", "XCoordinate",
            "Coordinate-X", "CoordinateX", "Position-X", "PositionX"
        ]
    )
    gname = pickvar(
        cut_ds,
        [
            "OpticalGeneration",
            "*OpticalGeneration*",
            "*Optical*Generation*"
        ]
    )

    if xname is None:
        raise RuntimeError(
            f"X coordinate not found in {cut_ds}: {safe_list_variables(cut_ds)}"
        )
    if gname is None:
        raise RuntimeError(
            f"OpticalGeneration not found in {cut_ds}: "
            f"{safe_list_variables(cut_ds)}"
        )

    x_raw = get_data(cut_ds, xname)
    g_raw = get_data(cut_ds, gname)
    x, g = sort_unique_average(x_raw, g_raw)

    m = (
        np.isfinite(x) & np.isfinite(g) &
        (x >= ACTIVE_X_MIN_UM) &
        (x <= ACTIVE_X_MAX_UM)
    )
    xa = x[m]
    ga = g[m]

    if xa.size < 2:
        raise RuntimeError(
            f"Too few active-aperture points at y={at_y}: {xa.size}"
        )

    return xa, ga
