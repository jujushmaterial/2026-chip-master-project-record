(define (make-box name material x0 y0 z0 x1 y1 z1)
  (sdegeo:create-cuboid
    (position x0 y0 z0)
    (position x1 y1 z1)
    material name))

(define (seg-name s) (car s))
(define (seg-lo s)   (cadr s))
(define (seg-hi s)   (caddr s))
(define (seg-type s) (cadddr s))
(define (seg-center s) (* 0.5 (+ (seg-lo s) (seg-hi s))))

(define Nseg (length ActiveSegments))

; Whole-block Candidate-B retention rule: keep a full Cartesian block only
; when its center lies inside the nominal 12.5-um active radius.
(define (kept-cell? i j)
  (and (>= i 0)
       (< i Nseg)
       (>= j 0)
       (< j Nseg)
       (let* ((sx (list-ref ActiveSegments i))
              (sy (list-ref ActiveSegments j))
              (cx (seg-center sx))
              (cy (seg-center sy)))
         (<= (sqrt (+ (* cx cx) (* cy cy))) R_active))))

(define ActiveSTIBodies (list))
(define UpperSiBodies   (list))

(define (make-active-cell i j)
  (let* ((sx (list-ref ActiveSegments i))
         (sy (list-ref ActiveSegments j))
         (x0 (seg-lo sx))
         (x1 (seg-hi sx))
         (y0 (seg-lo sy))
         (y1 (seg-hi sy))
         (base-name (string-append (seg-name sx) "." (seg-name sy))))
    (if (or (string=? (seg-type sx) "STI")
            (string=? (seg-type sy) "STI"))
        (let ((b
                (make-box
                  (string-append "R.ActiveSTI." base-name)
                  "Oxide"
                  x0 y0 z_STI_bottom x1 y1 z_uSi_top)))
          (set! ActiveSTIBodies
                (append ActiveSTIBodies (list b))))
        (begin
          (make-box
            (string-append "R.BOX." base-name)
            "Oxide"
            x0 y0 z_surface x1 y1 z_BOX_top)
          ; Electrical model follows the frozen 2D baseline: UpperSi is
          ; intentionally inactive Oxide for SDevice, not physical Silicon.
          (let ((u
                  (make-box
                    (string-append "R.UpperSi." base-name)
                    "Oxide"
                    x0 y0 z_BOX_top x1 y1 z_uSi_top)))
            (set! UpperSiBodies
                  (append UpperSiBodies (list u))))))))

; STI5 follows the retained Candidate-B staircase perimeter.
; - exposed side strips use the full 0.600-um principal-cut width so x=0/y=0
;   retains the frozen 12.20-12.80 um interval;
; - convex-corner fill uses only 0.300 um (= ov_STI5), preventing the diagonal
;   corner fill from entering the frozen cathode NW annulus (r>=14.20 um).
; This is a geometry-consistency realization of the surrogate STI5 topology;
; it is NOT claimed as an exact foundry mask.
(define STI5Bodies (list))

(define (make-sti5-box name x0 y0 x1 y1)
  (let ((body
          (make-box name "Oxide"
            x0 y0 z_STI_bottom x1 y1 z_uSi_top)))
    (set! STI5Bodies (append STI5Bodies (list body)))
    body))

(define (make-sti5-around-cell i j)
  (let* ((sx (list-ref ActiveSegments i))
         (sy (list-ref ActiveSegments j))
         (x0 (seg-lo sx))
         (x1 (seg-hi sx))
         (y0 (seg-lo sy))
         (y1 (seg-hi sy))
         (base-name (string-append (seg-name sx) "." (seg-name sy)))
         (L (not (kept-cell? (- i 1) j)))
         (R (not (kept-cell? (+ i 1) j)))
         (B (not (kept-cell? i (- j 1))))
         (T (not (kept-cell? i (+ j 1)))))
    (if L
        (make-sti5-box
          (string-append "R.STI5." base-name ".L")
          (- x0 w_STI5) y0 x0 y1))
    (if R
        (make-sti5-box
          (string-append "R.STI5." base-name ".R")
          x1 y0 (+ x1 w_STI5) y1))
    (if B
        (make-sti5-box
          (string-append "R.STI5." base-name ".B")
          x0 (- y0 w_STI5) x1 y0))
    (if T
        (make-sti5-box
          (string-append "R.STI5." base-name ".T")
          x0 y1 x1 (+ y1 w_STI5)))
    (if (and L B)
        (make-sti5-box
          (string-append "R.STI5." base-name ".LB")
          (- x0 w_STI5_corner) (- y0 w_STI5_corner) x0 y0))
    (if (and L T)
        (make-sti5-box
          (string-append "R.STI5." base-name ".LT")
          (- x0 w_STI5_corner) y1 x0 (+ y1 w_STI5_corner)))
    (if (and R B)
        (make-sti5-box
          (string-append "R.STI5." base-name ".RB")
          x1 (- y0 w_STI5_corner) (+ x1 w_STI5_corner) y0))
    (if (and R T)
        (make-sti5-box
          (string-append "R.STI5." base-name ".RT")
          x1 y1 (+ x1 w_STI5_corner) (+ y1 w_STI5_corner)))))

; Explicit annulus helper used only for the outer electrical pickup topology.
; STI6/STI7 exact 3D masks are unpublished; the annular realization is a
; surrogate chosen to preserve the frozen 2D cylindrical principal cut.
(define (make-annulus name material rin rout z0 z1)
  (sdegeo:set-default-boolean "XX")
  (let* ((outer
           (sdegeo:create-cylinder
             (position 0.0 0.0 z0)
             (position 0.0 0.0 z1)
             rout material name))
         (inner
           (sdegeo:create-cylinder
             (position 0.0 0.0 z0)
             (position 0.0 0.0 z1)
             rin material (string-append name ".CUT")))
         (ring (sdegeo:bool-subtract (list outer inner))))
    ; Explicit cleanup is harmless if bool-subtract already consumed the cutter.
    (let ((leftover (find-region-id (string-append name ".CUT"))))
      (if (not (null? leftover))
          (sdegeo:delete-region leftover)))
    (sde:add-material ring material name)
    (sdegeo:set-default-boolean "ABA")
    (remove-body-ABA ring)
    ring))


; =============================================================================
; 1A. Native 3D Ref/Eval helpers for robust electrical meshing
; =============================================================================
; Mesh-only windows must not modify physical geometry.  v2.0.1 therefore uses
; only documented Ref/Eval Cuboid / multi-Cuboid objects.  No sweep, cone,
; temporary physical body, Boolean mesh-solid, or extract-refpolyhedron is used.

(define (cuboid-shape x0 y0 z0 x1 y1 z1)
  (list "Cuboid"
    (position x0 y0 z0)
    (position x1 y1 z1)))

(define (define-cuboid-window name x0 y0 z0 x1 y1 z1)
  (sdedr:define-refeval-window name "Cuboid"
    (position x0 y0 z0)
    (position x1 y1 z1)))

; Conservative square-ring mesh envelope made from four native cuboids.
; This is numerical only and does not redefine the radial electrical core.
(define (square-ring-shapes rin rout z0 z1)
  (list
    (cuboid-shape   rin (- rout) z0   rout   rout z1)
    (cuboid-shape (- rout) (- rout) z0 (- rin) rout z1)
    (cuboid-shape (- rin)   rin z0    rin   rout z1)
    (cuboid-shape (- rin) (- rout) z0 rin (- rin) z1)))

; Active STI1-4 are Cartesian, not annular.  Build a compact multi-Cuboid
; refinement window from the actual STI x/y strip coordinates in ActiveSegments.
; There are 8 STI segments in X and the same 8 in Y -> 16 strip windows.
(define (build-active-sti-mesh-shapes)
  (let ((shapes (list)))
    (do ((i 0 (+ i 1)))
        ((>= i Nseg) shapes)
      (let ((s (list-ref ActiveSegments i)))
        (if (string=? (seg-type s) "STI")
            (begin
              ; vertical strip following the physical Cartesian STI network
              (set! shapes
                (append shapes
                  (list
                    (cuboid-shape
                      (- (seg-lo s) 0.10) (- R_active) -0.18
                      (+ (seg-hi s) 0.10)    R_active (+ z_uSi_top 0.020)))))
              ; horizontal strip following the physical Cartesian STI network
              (set! shapes
                (append shapes
                  (list
                    (cuboid-shape
                      (- R_active) (- (seg-lo s) 0.10) -0.18
                         R_active  (+ (seg-hi s) 0.10) (+ z_uSi_top 0.020)))))))))))

(define (define-frozen-doping)

    "Place.Substrate" "Dop.Substrate" "Bulk")

; 5.2 Mesh-only 3D CYLINDER / ANNULUS Ref/Eval helper functions

; Snapshot the physical body count. No temporary body is allowed to remain.
(define V26_MeshBodyCountBefore (length (get-body-list)))

(define (v26-cylinder-refwin rwname tmpname radius z0 z1)
  (sdegeo:set-default-boolean "XX")
  (let ((tmp (sdegeo:create-cylinder
               (position 0.0 0.0 z0) (position 0.0 0.0 z1)
               radius "Silicon" tmpname)))
    (extract-refpolyhedron tmp rwname)
    (sdegeo:delete-region tmp))
  (sdegeo:set-default-boolean "ABA"))

(define (v26-annulus-refwin rwname tmpbase rin rout z0 z1)
  (sdegeo:set-default-boolean "XX")
  (let* ((outer (sdegeo:create-cylinder
                  (position 0.0 0.0 z0) (position 0.0 0.0 z1)
                  rout "Silicon" (string-append tmpbase ".OUT")))
         (inner (sdegeo:create-cylinder
                  (position 0.0 0.0 z0) (position 0.0 0.0 z1)
                  rin "Silicon" (string-append tmpbase ".IN")))
         (ring (sdegeo:bool-subtract (list outer inner))))
    (extract-refpolyhedron ring rwname)
    (sdegeo:delete-region ring)
    ; Boolean subtraction normally consumes the cutter. Clean up only if not.
    (let ((residual-in (find-region-id (string-append tmpbase ".IN")))
          (residual-out (find-region-id (string-append tmpbase ".OUT"))))
      (if (not (null? residual-in)) (sdegeo:delete-region residual-in))
      (if (not (null? residual-out)) (sdegeo:delete-region residual-out))))
  (sdegeo:set-default-boolean "ABA"))
