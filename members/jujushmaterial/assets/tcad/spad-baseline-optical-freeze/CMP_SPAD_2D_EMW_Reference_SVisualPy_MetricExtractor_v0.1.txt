CASE_ID = "Reference_2D_940nm_TE"

# Representative optical/SCR anchor from frozen electrical baseline.
Y_ANCHOR_UM = 0.180202945597

# ~25 um reference SPAD active diameter -> +/-12.5 um lateral aperture.
ACTIVE_X_MIN_UM = -12.5
ACTIVE_X_MAX_UM = +12.5

PREFIX = f"n{VIS_NODE}_emw_ref_metric_v01"
FULL_CSV = f"{PREFIX}_anchor_full.csv"

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------
def is_svisual_error(obj):
    if not isinstance(obj, str):
        return False
    s = obj.strip().lower()
    return (
        s.startswith("error:") or
        s.startswith("validation failed") or
        "reference resolving failed" in s or
        "does not exist" in s
    )


def safe_list_variables(dataset):
    result = sv.list_variables(dataset=dataset)
    if is_svisual_error(result):
        raise RuntimeError(
            f"{dataset}: list_variables returned SVisual error: {result}"
        )
    if isinstance(result, str):
        return [result]
    return list(result)


def pickvar(dataset, patterns):
    names = safe_list_variables(dataset)
    for pat in patterns:
        p = pat.lower()
        for name in names:
            if fnmatch.fnmatch(name.lower(), p):
                return name
    return None


def get_data(dataset, varname):
    return np.asarray(
        sv.get_variable_data(varname=varname, dataset=dataset),
        dtype=float
    )


def finite_xy(x, y):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)
    m = np.isfinite(x) & np.isfinite(y)
    return x[m], y[m]


def sort_unique_average(x, y):
    x, y = finite_xy(x, y)
    idx = np.argsort(x, kind="mergesort")
    x = x[idx]
    y = y[idx]

    xu, inv = np.unique(x, return_inverse=True)
    sums = np.bincount(inv, weights=y)
    counts = np.bincount(inv)

    yu = sums / np.maximum(counts, 1)
    return xu, yu


def save_csv(filename, columns):
    names = list(columns.keys())
    arrays = [np.asarray(columns[name], dtype=float) for name in names]

    if not arrays:
        raise RuntimeError(f"{filename}: no columns")

    n = len(arrays[0])
    if any(len(a) != n for a in arrays):
        raise RuntimeError(
            f"{filename}: column-length mismatch "
            f"{[len(a) for a in arrays]}"
        )

    data = np.column_stack(arrays)
    np.savetxt(
        filename,
        data,
        delimiter=",",
        header=",".join(names),
        comments="",
        fmt="%.12e"
    )


def metric_stats(values):
    v = np.asarray(values, dtype=float)
    v = v[np.isfinite(v)]

    if v.size == 0:
        raise RuntimeError("No finite OpticalGeneration values in metric window")

    mean = float(np.mean(v))
    std = float(np.std(v))

    return {
        "count": int(v.size),
        "mean": mean,
        "median": float(np.median(v)),
        "min": float(np.min(v)),
        "max": float(np.max(v)),
        "std": std,
        "cv": float(std / mean) if mean != 0.0 else np.nan,
        "p10": float(np.percentile(v, 10.0)),
        "p90": float(np.percentile(v, 90.0)),
        "max_over_mean": float(np.max(v) / mean) if mean != 0.0 else np.nan,
    }


def write_metric(fh, name, value):
    if value is None:
        fh.write(f"{name} = NA\n")
        return

    try:
        fv = float(value)
    except Exception:
        fh.write(f"{name} = {value}\n")
        return

    if np.isfinite(fv):
        fh.write(f"{name} = {fv:.12g}\n")
    else:
        fh.write(f"{name} = NA\n")


def create_horizontal_cut(source_plot, source_ds, at_y, name):
    attempts = [
        (
            "plot+Silicon",
            dict(
                at=at_y,
                plot=source_plot,
                materials=["Silicon"],
                name=name
            )
        ),
        (
            "plot-only",
            dict(
                at=at_y,
                plot=source_plot,
                name=name
            )
        ),
        (
            "dataset+Silicon",
            dict(
                at=at_y,
                dataset=source_ds,
                materials=["Silicon"],
                name=name
            )
        ),
        (
            "dataset-only",
            dict(
                at=at_y,
                dataset=source_ds,
                name=name
            )
        ),
    ]

    errors = []

    for label, kwargs in attempts:
        try:
            result = sv.create_cutline("y", **kwargs)
        except Exception as exc:
            errors.append(f"{label}: exception={exc}")
            continue

        if is_svisual_error(result):
            errors.append(f"{label}: returned={result}")
            continue

        print(
            f"EMW_METRIC_V01: CUT_CREATED method={label} "
            f"dataset={result}"
        )
        return result

    raise RuntimeError(
        "Anchor horizontal cutline creation failed: "
        + " || ".join(errors)
    )


# ----------------------------------------------------------------------
# 1. Load the EMW optical-generation TDR
# ----------------------------------------------------------------------
print(f"EMW_METRIC_V01: CASE={CASE_ID}")
print(f"EMW_METRIC_V01: INPUT={OPTICAL_TDR}")

if not os.path.exists(OPTICAL_TDR):
    raise RuntimeError(
        f"EMW optical result does not exist: {OPTICAL_TDR}"
    )

requested_ds = "EMW_OPTICAL"
loaded_ds = sv.load_file(OPTICAL_TDR, name=requested_ds)

if is_svisual_error(loaded_ds):
    raise RuntimeError(
        f"load_file returned SVisual error: {loaded_ds}"
    )

source_ds = (
    loaded_ds
    if isinstance(loaded_ds, str) and loaded_ds.strip()
    else requested_ds
)

print(
    f"EMW_METRIC_V01: LOAD requested={requested_ds} "
    f"returned={loaded_ds} using_dataset={source_ds}"
)

print(
    f"EMW_METRIC_V01: SOURCE_VARS="
    f"{safe_list_variables(source_ds)}"
)

# Reproduce the already proven T-2022.03 SVisualPy flow:
# load_file -> create 2D plot -> create cutline from plot.
source_plot = sv.create_plot(
    name="Plot_EMW_OpticalGeneration_2D",
    dataset=source_ds
)

if is_svisual_error(source_plot):
    raise RuntimeError(
        f"create_plot returned SVisual error: {source_plot}"
    )

# ----------------------------------------------------------------------
# 2. Horizontal cut at representative y ~= 0.180203 um
# ----------------------------------------------------------------------
cut_ds = create_horizontal_cut(
    source_plot=source_plot,
    source_ds=source_ds,
    at_y=Y_ANCHOR_UM,
    name="CUT_ANCHOR_0180"
)

print(
    f"EMW_METRIC_V01: CUT_VARS="
    f"{safe_list_variables(cut_ds)}"
)

xname = pickvar(
    cut_ds,
    [
        "X", "x",
        "X-Coordinate", "XCoordinate",
        "Coordinate-X", "CoordinateX",
        "Position-X", "PositionX"
    ]
)

gname = pickvar(
    cut_ds,
    [
        "OpticalGeneration",
        "*OpticalGeneration*",
        "*Optical*Generation*"
    ]
)

if xname is None:
    raise RuntimeError(
        f"X coordinate not found in anchor cutline: "
        f"{safe_list_variables(cut_ds)}"
    )

if gname is None:
    raise RuntimeError(
        f"OpticalGeneration variable not found in anchor cutline: "
        f"{safe_list_variables(cut_ds)}"
    )

x_raw = get_data(cut_ds, xname)
g_raw = get_data(cut_ds, gname)

x, g = sort_unique_average(x_raw, g_raw)

if x.size < 2:
    raise RuntimeError(
        f"Anchor cutline contains too few points: {x.size}"
    )

# ----------------------------------------------------------------------
# 3. Save full cutline and active-aperture subset
# ----------------------------------------------------------------------
save_csv(
    FULL_CSV,
    {
        "X_um": x,
        "OpticalGeneration_cm-3_s-1": g,
    }
)

m_active = (
    np.isfinite(x) &
    np.isfinite(g) &
    (x >= ACTIVE_X_MIN_UM) &
    (x <= ACTIVE_X_MAX_UM)
)

xa = x[m_active]
ga = g[m_active]

if xa.size < 2:
    raise RuntimeError(
        "No usable points inside active aperture "
        f"[{ACTIVE_X_MIN_UM}, {ACTIVE_X_MAX_UM}] um"
    )

save_csv(
    ACTIVE_CSV,
    {
        "X_um": xa,
        "OpticalGeneration_cm-3_s-1": ga,
    }
)

stats_full = metric_stats(g)
stats_active = metric_stats(ga)

# ----------------------------------------------------------------------
# 4. Write metrics
# ----------------------------------------------------------------------
with open(METRICS_TXT, "w") as fh:
    fh.write("# CMP SPAD 2D EMW Reference Metric Extractor v0.1\n")
    fh.write("# Diagnostic-only quantitative baseline step\n")
    fh.write("# Final G_SCR is NOT computed in this version.\n\n")

    fh.write(f"CaseID = {CASE_ID}\n")
    fh.write("ReferencePitch = N/A\n")
    fh.write("ReferenceFF = N/A\n")
    write_metric(fh, "AnchorDepth_um", Y_ANCHOR_UM)
    write_metric(fh, "ActiveXMin_um", ACTIVE_X_MIN_UM)
    write_metric(fh, "ActiveXMax_um", ACTIVE_X_MAX_UM)

    fh.write("\n[FullAnchorCutline]\n")
    for key, val in stats_full.items():
        write_metric(fh, f"Ganchor_full_{key}", val)

    fh.write("\n[ActiveApertureAnchorCutline]\n")
    for key, val in stats_active.items():
        write_metric(fh, f"Ganchor_active_{key}", val)

    fh.write("\n[Files]\n")
    fh.write(f"FullCSV = {FULL_CSV}\n")
    fh.write(f"ActiveCSV = {ACTIVE_CSV}\n")
