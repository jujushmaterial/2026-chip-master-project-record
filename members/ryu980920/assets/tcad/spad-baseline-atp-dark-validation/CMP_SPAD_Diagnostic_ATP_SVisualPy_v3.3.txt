def pickvar(dataset, patterns):
    names = list(sv.list_variables(dataset=dataset))
    for pat in patterns:
        p = pat.lower()
        for name in names:
            if fnmatch.fnmatch(name.lower(), p):
                return name
    return None


def get_data(dataset, varname):

def finite_xy(x, y):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)
    m = np.isfinite(x) & np.isfinite(y)
    return x[m], y[m]


def sort_unique_many(x, *ys):
    x = np.asarray(x, dtype=float)
    arrs = [np.asarray(y, dtype=float) for y in ys]

    m = np.isfinite(x)
    for a in arrs:
        m &= np.isfinite(a)

    x = x[m]
    arrs = [a[m] for a in arrs]

    idx = np.argsort(x, kind="mergesort")
    x = x[idx]
    arrs = [a[idx] for a in arrs]

    xu, inv = np.unique(x, return_inverse=True)
    outs = []
    for a in arrs:
        sums = np.bincount(inv, weights=a)
        counts = np.bincount(inv)
        outs.append(sums / np.maximum(counts, 1))
    return (xu, *outs)


def sort_unique_with_optional(x, main, optional_arrays):
    x = np.asarray(x, dtype=float)
    main = np.asarray(main, dtype=float)

    mask = np.isfinite(x) & np.isfinite(main)
    x = x[mask]
    main = main[mask]

    opts = []
    for arr in optional_arrays:
        if arr is None:
            opts.append(None)
        else:
            arr = np.asarray(arr, dtype=float)
            opts.append(arr[mask])

    order = np.argsort(x, kind="mergesort")
    x = x[order]
    main = main[order]
    opts = [None if arr is None else arr[order] for arr in opts]

    xu, inv = np.unique(x, return_inverse=True)

    main_sum = np.bincount(inv, weights=main)
    counts = np.bincount(inv)
    main_u = main_sum / np.maximum(counts, 1)

    opt_u = []
    for arr in opts:
        if arr is None:
            opt_u.append(np.full_like(xu, np.nan, dtype=float))
            continue

        good = np.isfinite(arr)
        sums = np.bincount(inv[good], weights=arr[good], minlength=len(xu))
        cnts = np.bincount(inv[good], minlength=len(xu))
        out = np.full(len(xu), np.nan, dtype=float)
        valid = cnts > 0
        out[valid] = sums[valid] / cnts[valid]
        opt_u.append(out)

    return (xu, main_u, *opt_u)


def crossings(x, y, level, lo=-np.inf, hi=np.inf):
    x, y = finite_xy(x, y)
    if x.size < 2:
        return np.array([], dtype=float)

    idx = np.argsort(x, kind="mergesort")
    x = x[idx]
    y = y[idx]

    x1 = x[:-1]
    x2 = x[1:]
    y1 = y[:-1]
    y2 = y[1:]

    m = (
        (y2 != y1) &
        ((y1 - level) * (y2 - level) <= 0.0)
    )

    if not np.any(m):
        return np.array([], dtype=float)

    t = (level - y1[m]) / (y2[m] - y1[m])
    good_t = (t >= 0.0) & (t <= 1.0)

    xc = x1[m][good_t] + t[good_t] * (x2[m][good_t] - x1[m][good_t])

    return xc[(xc >= lo) & (xc <= hi)]


def bracket_crossings(vals, x0):
    vals = np.asarray(vals, dtype=float)
    vals = np.sort(vals[np.isfinite(vals)])
    lower = vals[vals <= x0]
    upper = vals[vals > x0]
    lo = float(lower[-1]) if lower.size else np.nan
    hi = float(upper[0]) if upper.size else np.nan
    return lo, hi


def window_max(x, y, lo, hi, positive_only=False):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)

    m = np.isfinite(x) & np.isfinite(y) & (x >= lo) & (x <= hi)
    if positive_only:
        m &= (y > 0.0)

    if not np.any(m):
        return np.nan, np.nan

    xx = x[m]
    yy = y[m]
    k = int(np.argmax(yy))
    return float(yy[k]), float(xx[k])


def vbd_from_phi(v, pe, ph):
    v, pe, ph = sort_unique_many(v, pe, ph)
    g = np.minimum(pe, ph)
    c = crossings(v, g, 1.0)
    return float(c[0]) if c.size else np.nan


def field_mag(dataset):
    scalar = pickvar(
        dataset,
        ["AbsElectricField", "Abs(ElectricField-V)", "ElectricField"]
    )
    if scalar is not None:
        a = get_data(dataset, scalar)
        if a.ndim == 1:
            return np.abs(a)

    exn = pickvar(
        dataset,
        ["ElectricField-X", "ElectricField/x", "*ElectricField*X*"]
    )
    eyn = pickvar(
        dataset,
        ["ElectricField-Y", "ElectricField/y", "*ElectricField*Y*"]
    )
    if exn is None or eyn is None:
        raise RuntimeError("Electric-field variable not found")

    ex = get_data(dataset, exn)
    ey = get_data(dataset, eyn)
    return np.sqrt(ex * ex + ey * ey)


def qn_ratio_sorted(z_if, p, e):
    z_if = np.asarray(z_if, dtype=float)
    p = np.asarray(p, dtype=float)
    e = np.asarray(e, dtype=float)

    if z_if.size < 3:
        return np.zeros_like(z_if)

    z_cm = z_if * 1.0e-4
    dpdz = np.gradient(p, z_cm, edge_order=1)
    ratio = np.zeros_like(z_if)

    good = (
        np.isfinite(p) & np.isfinite(e) & np.isfinite(dpdz) &
        (p > 0.0) & (e > 0.0) &
        (np.abs(dpdz) > 0.0)
    )
    ratio[good] = np.abs(p[good] * e[good] / (VT * dpdz[good]))
    return ratio


def first_ge_after(x, y, x0, level, xhi):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)

    for i in range(1, len(x)):
        if x[i] < x0:
            continue
        if x[i] > xhi:
            break

        if np.isfinite(y[i]) and y[i] >= level:
            xp, yp = x[i - 1], y[i - 1]
            if (
                xp >= x0 and np.isfinite(yp) and
                yp < level and y[i] != yp
            ):
                t = (level - yp) / (y[i] - yp)
                return float(xp + t * (x[i] - xp))
            return float(x[i])

    return np.nan


def emit(fh, name, value):
    if value is None or not np.isfinite(value):
        fh.write(f"{name} = NA\n")
        print(f"DOE: {name} x")
    else:
        value = float(value)
        fh.write(f"{name} = {value:.12g}\n")
        print(f"DOE: {name} {value:.12g}")


def write_metric(fh, name, value):
    if value is None or not np.isfinite(value):
        fh.write(f"{name} = NA\n")
    else:
        fh.write(f"{name} = {float(value):.12g}\n")


def save_csv(filename, columns):
    names = list(columns.keys())
    arrays = [np.asarray(columns[name], dtype=float) for name in names]

    if len(arrays) == 0:
        return

    n = len(arrays[0])
    for a in arrays:
        if len(a) != n:
            raise RuntimeError(
                f"CSV column length mismatch in {filename}: "
                f"{[len(x) for x in arrays]}"
            )

    data = np.column_stack(arrays)
    np.savetxt(
        filename,
        data,
        delimiter=",",
        header=",".join(names),
        comments="",
        fmt="%.12e"
    )

def snapshot_bias(path):
    name = os.path.basename(path)
    match = regex.match(rf"n{SNODE}_V(\d{{3}})_des.tdr$", name)
    if match is None:
        return np.nan
    return float(int(match.group(1))) / 10.0


def choose_atp_snapshot(vbd_value):
    paths = sorted(glob.glob(f"n{SNODE}_V???_des.tdr"))
    candidates = []
    for path in paths:
        bias = snapshot_bias(path)
        if np.isfinite(bias):
            candidates.append((bias, path))

    if not candidates:
        return np.nan, None

    candidates.sort(key=lambda item: item[0])

    if np.isfinite(vbd_value):
        target = float(vbd_value) + ATP_TARGET_VEX
        eligible = [item for item in candidates if item[0] >= vbd_value]
        if not eligible:
            return np.nan, None
        above_target = [item for item in eligible if item[0] >= target]
        return above_target[0] if above_target else eligible[-1]

    return min(candidates, key=lambda item: abs(item[0] - ATP_FALLBACK_BIAS))


def interp_inside(x, y, x0):
    x, y = sort_unique_many(x, y)
    if x.size < 2 or x0 < x[0] or x0 > x[-1]:
        return np.nan
    return float(np.interp(x0, x, y))

# T-2022.03 workflow:
#   - force all ATP TDR fields to load when the Python wrapper supports it,
#   - create an explicit 2D ATP plot,
#   - create 1D cutlines from that plot,
#   - reconstruct a sampled cylindrical r-z ATP map.
#
# Research-facing ATP DOE definitions:
#   ATP_axis180   = jBreakdownProbability at r=0, z=180 nm
#   ATP_peak180   = radial maximum of jBreakdownProbability at z=180 nm
#   ATP_peak      = global maximum over the sampled (r,z) ATP map
#   ATP_peak_r_um = radial position of ATP_peak
#   ATP_peak_z_nm = depth position of ATP_peak from the grating/Si interface
#
# Legacy r=4.6875 um and r=12.5 um profiles are still exported as
# diagnostic CSVs, but they are no longer used as the main ATP DOE metrics.

# ATP_MAP_CSV is a sampled cylindrical r-z map, not an original TDR-node dump.

def create_atp_cut(source_plot, radius, name):
    try:
        return sv.create_cutline(
            "x",
            at=radius,
            plot=source_plot,
            name=name
        )

    except TypeError:
        return sv.create_cutline(
            type="x",
            at=radius,
            plot=source_plot,
            name=name
        )


def safe_list_variables(dataset):
    try:
        return list(
            sv.list_variables(
                dataset=dataset
            )
        )
    except Exception as exc:
        print(
            "SVISUALPY_STAGE1: ATP_LIST_VARIABLES_WARNING "
            f"dataset={dataset}: {exc}"
        )
        return []


def probe_exact_variable(dataset, candidates):
    names = safe_list_variables(dataset)

    for candidate in candidates:
        for name in names:
            if name.lower() == candidate.lower():
                return name

    for candidate in candidates:
        try:
            arr = get_data(
                dataset,
                candidate
            )
            if arr.size > 0:
                return candidate
        except Exception:
            pass

    return None


def pick_cut_variable(dataset, patterns, exact_candidates):
    names = safe_list_variables(dataset)

    for pat in patterns:
        p = pat.lower()
        for name in names:
            if fnmatch.fnmatch(
                name.lower(),
                p
            ):
                return name

    return probe_exact_variable(
        dataset,
        exact_candidates
    )


def read_atp_cut(source_plot, radius, name, verbose=False):
    try:
        cut = create_atp_cut(
            source_plot,
            radius,
            name
        )
    except Exception as exc:
        if verbose:
            print(
                "SVISUALPY_STAGE1: ATP_CUT_FAIL "
                f"r={radius:.6g}: {exc}"
            )
        return None

    names = safe_list_variables(cut)

    if verbose:
        print(
            "SVISUALPY_STAGE1: ATP_CUT_DATASET "
            f"r={radius:.6g} dataset={cut}"
        )
        print(
            "SVISUALPY_STAGE1: ATP_CUT_VARIABLES "
            f"r={radius:.6g} names={names}"
        )

    y_name = pick_cut_variable(
        cut,
        ["Y", "y", "*Y*"],
        ["Y", "y"]
    )

    j_name = pick_cut_variable(
        cut,
        [
            "jBreakdownProbability",
            "*jBreakdownProbability*",
            "*Joint*BreakdownProbability*"
        ],
        [
            "jBreakdownProbability",
            "JBreakdownProbability"
        ]
    )

    e_name = pick_cut_variable(
        cut,
        [
            "eBreakdownProbability",
            "*eBreakdownProbability*"
        ],
        [
            "eBreakdownProbability",
            "EBreakdownProbability"
        ]
    )

    h_name = pick_cut_variable(
        cut,
        [
            "hBreakdownProbability",
            "*hBreakdownProbability*"
        ],
        [
            "hBreakdownProbability",
            "HBreakdownProbability"
        ]
    )

    if y_name is None or j_name is None:
        if verbose:
            print(
                "SVISUALPY_STAGE1: ATP_CUT_VARIABLE_MISSING "
                f"r={radius:.6g} "
                f"Y={y_name} J={j_name} "
                f"E={e_name} H={h_name}"
            )
        return None

    try:
        y_raw = get_data(
            cut,
            y_name
        )
        pj_raw = get_data(
            cut,
            j_name
        )
    except Exception as exc:
        if verbose:
            print(
                "SVISUALPY_STAGE1: ATP_CUT_DATA_READ_FAIL "
                f"r={radius:.6g}: {exc}"
            )
        return None

    if len(y_raw) != len(pj_raw):
        if verbose:
            print(
                "SVISUALPY_STAGE1: ATP_CUT_LENGTH_MISMATCH "
                f"r={radius:.6g} "
                f"Y={len(y_raw)} J={len(pj_raw)}"
            )
        return None

    z_raw = (
        y_raw -
        Y_GRATING_SI_INTERFACE
    )

    pe_raw = None
    ph_raw = None

    if e_name is not None:
        try:
            candidate = get_data(
                cut,
                e_name
            )
            if len(candidate) == len(y_raw):
                pe_raw = candidate
        except Exception:
            pe_raw = None

    if h_name is not None:
        try:
            candidate = get_data(
                cut,
                h_name
            )
            if len(candidate) == len(y_raw):
                ph_raw = candidate
        except Exception:
            ph_raw = None

    z, pj, pe, ph = sort_unique_with_optional(
        z_raw,
        pj_raw,
        [pe_raw, ph_raw]
    )

    finite = (
        np.isfinite(z) &
        np.isfinite(pj)
    )

    if not np.any(finite):
        if verbose:
            print(
                "SVISUALPY_STAGE1: ATP_CUT_NO_FINITE_J "
                f"r={radius:.6g}"
            )
        return None

    z = z[finite]
    pj = pj[finite]
    pe = pe[finite]
    ph = ph[finite]

    return {
        "cut": cut,
        "joint_name": j_name,
        "electron_name": e_name,
        "hole_name": h_name,
        "z": z,
        "pj": pj,
        "pe": pe,
        "ph": ph,
    }


if atp_file is None or not os.path.isfile(atp_file):
