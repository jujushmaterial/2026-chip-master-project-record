*=============================================================================
* CMP SPAD diagnostic deck: ionization integrals + breakdown paths + ATP
* Version 1.0, 2026-08-27
*
* Source merge
*   - Workbench dependency, contacts, voltage landmarks, and file naming:
*     teammate VBD diagnostic deck v1.1
*   - Ionization-path controls: user deck B2
*   - BreakdownProbability and ATP fields: user deck B3
*   - Dark-generation model card: user decks B1/B3
*
* PHYSICS MODE: IMPACT-IONIZATION FEEDBACK OFF
*   AvalPostProcessing excludes impact-ionization-generated carriers from the
*   device continuity equations.  This makes it possible to continue beyond
*   breakdown without current divergence and to obtain ATP above VBD.
*
* Deliberate integration decisions
*   1. BreakAtIonIntegral is NOT used.  The ramp must continue to 20 V so that
*      above-breakdown ATP snapshots exist.
*   2. ElectricField is the avalanche driving force.  This is chosen for
*      consistency with approximate breakdown-path integration.  The separate
*      feedback-ON I-V deck uses Eparallel; therefore a VBD difference between
*      the two decks contains a driving-force-model contribution as well as a
*      numerical/definition contribution.
*   3. The first run must verify that PhiElectron and PhiHole are present in
*      the current plot file.  The supplied SVisualPy extracts VBD from their
*      crossing of one.
*=============================================================================

}

Math {
  Cylindrical
  Extrapolate
  Derivatives
  RelErrControl
  Digits             = 6
  ErrRef( Electron ) = 1.0e4
  ErrRef( Hole )     = 1.0e4
  Iterations         = 100
  Notdamped          = 100
  Number_of_Threads  = 4
  RhsMin             = 1.0e-15
  ExitOnFailure

  AvalPostProcessing

  ComputeIonizationIntegrals(
    MinElectricField      = 2.0e5
    SaveEIonIntegralPaths = 1
    SaveHIonIntegralPaths = 1
    WriteAll
  )
}

Solve {
  Coupled( Iterations = 200 LineSearchDamping = 1.0e-4 ) { Poisson }
  Coupled { Poisson Electron Hole }

  Plot( FilePrefix = "n@node@_V000" )

  * Fast approach to 14 V, still below the expected 15.8 V anchor.
  Quasistationary(
    InitialStep = 1.0e-3
    MinStep     = 1.0e-8
    MaxStep     = 2.0e-2
    Increment   = 1.35
    Decrement   = 2.0
    Goal { Name = "cathode" Voltage = 14.0 }
  ){
    Coupled { Poisson Electron Hole }
    CurrentPlot( Time = ( Range = (0 1) Intervals = 140 ) )
  }

  Plot( FilePrefix = "n@node@_V140" )

  * No BreakAtIonIntegral: continue through VBD to collect ATP above VBD.
  Quasistationary(
    InitialStep = 1.0e-3
    MinStep     = 1.0e-9
    MaxStep     = 2.0e-3
    Increment   = 1.25
    Decrement   = 2.0
    Goal { Name = "cathode" Voltage = 20.0 }
  ){
    Coupled { Poisson Electron Hole }
