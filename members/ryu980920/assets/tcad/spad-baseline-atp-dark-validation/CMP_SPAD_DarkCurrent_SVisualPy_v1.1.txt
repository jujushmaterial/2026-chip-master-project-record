def pickvar(dataset, patterns):
    names = list(sv.list_variables(dataset=dataset))
    for pat in patterns:
        p = pat.lower()
        for name in names:
            if fnmatch.fnmatch(name.lower(), p):
                return name
    return None


def get_data(dataset, varname):
    return np.asarray(
        sv.get_variable_data(varname=varname, dataset=dataset),
        dtype=float
    )


def sort_unique_many(x, *ys):
    x = np.asarray(x, dtype=float)
    arrs = [np.asarray(y, dtype=float) for y in ys]

    mask = np.isfinite(x)
    for arr in arrs:
        mask &= np.isfinite(arr)

    x = x[mask]
    arrs = [arr[mask] for arr in arrs]
    order = np.argsort(x, kind="mergesort")
    x = x[order]
    arrs = [arr[order] for arr in arrs]

    xu, inv = np.unique(x, return_inverse=True)
    outputs = []
    for arr in arrs:
        sums = np.bincount(inv, weights=arr)
        counts = np.bincount(inv)
        outputs.append(sums / np.maximum(counts, 1))
    return (xu, *outputs)


def optional_on_grid(dataset, patterns, vraw, vgrid, absolute=True):
    name = pickvar(dataset, patterns)
    if name is None:
        return None, np.full_like(vgrid, np.nan)

    arr = get_data(dataset, name)
    if absolute:
        arr = np.abs(arr)
    vx, ax = sort_unique_many(vraw, arr)
    if vx.size < 2:
        return name, np.full_like(vgrid, np.nan)
    return name, np.interp(vgrid, vx, ax, left=np.nan, right=np.nan)


def crossings(x, y, level):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)
    output = []
    for i in range(len(y) - 1):
        a, b = y[i], y[i + 1]
        if not (np.isfinite(a) and np.isfinite(b)) or a == b:
            continue
        if (a >= level) == (b >= level):
            continue
        t = (level - a) / (b - a)
        output.append(x[i] + t * (x[i + 1] - x[i]))
    return output


def vbd_rule1(v, current, criterion):
    values = crossings(v, np.abs(current), criterion)
    return float(values[-1]) if values else np.nan


def fit_primary(v, i_on, v_fit_max):
    mask = (
        np.isfinite(v) & np.isfinite(i_on) &
        (v > 0.2) & (v <= v_fit_max) & (np.abs(i_on) > 0.0)
    )
    if np.count_nonzero(mask) < 5:
        return None, np.nan

    exponent, ln_a = np.polyfit(
        np.log(v[mask] + PRIM_VBI),
        np.log(np.abs(i_on[mask])),
        1
    )
    primary = np.exp(ln_a) * (v + PRIM_VBI) ** exponent
    return primary, float(exponent)


def vbd_rule2(v, i_on, i_primary, lo=0.05, hi=0.40):
    i_on = np.abs(i_on)
    i_primary = np.abs(i_primary)
    valid = (i_on > 0.0) & (i_primary > 0.0)
    inv_m = np.full_like(i_on, np.nan)
    inv_m[valid] = i_primary[valid] / i_on[valid]

    selected = np.isfinite(inv_m) & (inv_m >= lo) & (inv_m <= hi)
    if np.count_nonzero(selected) < 3:
        return np.nan

    slope, intercept = np.polyfit(v[selected], inv_m[selected], 1)
    return float(-intercept / slope) if slope != 0.0 else np.nan


def inverse_fit_point_count(i_on, i_primary, lo=0.05, hi=0.40):
    i_on = np.abs(np.asarray(i_on, dtype=float))
    i_primary = np.abs(np.asarray(i_primary, dtype=float))

    valid = (
        np.isfinite(i_on) &
        np.isfinite(i_primary) &
        (i_on > 0.0) &
        (i_primary > 0.0)
    )

    inv_m = np.full_like(i_on, np.nan)
    inv_m[valid] = i_primary[valid] / i_on[valid]

    selected = np.isfinite(inv_m) & (inv_m >= lo) & (inv_m <= hi)
    return int(np.count_nonzero(selected))


def vbd_selfcontained(v, i_on, iterations=3):
    v_fit = 8.0
    vbd = np.nan
    exponent = np.nan
    primary = None

    for _ in range(iterations):
        primary, exponent = fit_primary(v, i_on, v_fit)
        if primary is None:
            return np.nan, exponent, v_fit, None
        vbd = vbd_rule2(v, i_on, primary)
        if not np.isfinite(vbd):
            return vbd, exponent, v_fit, primary
        v_fit = max(3.0, 0.5 * vbd)

    primary, exponent = fit_primary(v, i_on, v_fit)
    if primary is not None:
        vbd = vbd_rule2(v, i_on, primary)
    return vbd, exponent, v_fit, primary


def vbd_from_phi(v, pe, ph):
    v, pe, ph = sort_unique_many(v, pe, ph)
    if v.size < 2:
        return np.nan

    g = np.minimum(pe, ph)
    values = crossings(v, g, 1.0)
    return float(values[0]) if values else np.nan


def load_diagnostic_vbd():
    if not os.path.isfile(DIAG_IV_FILE):
        print(f"SVISUALPY_DARK: DIAG_IV_MISSING={DIAG_IV_FILE}")
        return np.nan

    sv.load_file(DIAG_IV_FILE, name="DIAGIV")

    vdiag_name = pickvar(
        "DIAGIV",
        ["*cathode*OuterVoltage*", "*cathode*InnerVoltage*"]
    )
    pe_name = pickvar(
        "DIAGIV",
        ["PhiElectron", "*PhiElectron*"]
    )
    ph_name = pickvar(
        "DIAGIV",
        ["PhiHole", "*PhiHole*"]
    )

    if vdiag_name is None or pe_name is None or ph_name is None:
        print("SVISUALPY_DARK: DIAGNOSTIC_VBD_VARIABLE_MISSING")
        return np.nan

    vdiag = np.abs(get_data("DIAGIV", vdiag_name))
    pe = get_data("DIAGIV", pe_name)
    ph = get_data("DIAGIV", ph_name)

    return vbd_from_phi(vdiag, pe, ph)


def interp_inside(x, y, target):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)
    valid = np.isfinite(x) & np.isfinite(y)
    x = x[valid]
    y = y[valid]
    if x.size < 2 or target < x.min() or target > x.max():
        return np.nan
    order = np.argsort(x, kind="mergesort")
    return float(np.interp(target, x[order], y[order]))


def emit(file_handle, name, value):
    if value is None or not np.isfinite(value):
        file_handle.write(f"{name} = NA\n")
        print(f"DOE: {name} x")
    else:
        value = float(value)
        file_handle.write(f"{name} = {value:.12g}\n")
        print(f"DOE: {name} {value:.12g}")


def write_metric(file_handle, name, value):
    if value is None or not np.isfinite(value):
        file_handle.write(f"{name} = NA\n")
    else:
        file_handle.write(f"{name} = {float(value):.12g}\n")


def save_csv(filename, columns):
    names = list(columns.keys())
    arrays = [np.asarray(columns[name], dtype=float) for name in names]
    if not arrays:
        return
    lengths = [len(arr) for arr in arrays]
    if len(set(lengths)) != 1:
        raise RuntimeError(f"CSV length mismatch in {filename}: {lengths}")
    np.savetxt(
        filename,
        np.column_stack(arrays),
        delimiter=",",
        header=",".join(names),
        comments="",
        fmt="%.12e"
    )


def safe_plot(callable_obj, tag):
    try:
        return callable_obj()
    except Exception as exc:
        print(f"SVISUALPY_DARK: PLOT_WARNING[{tag}]: {exc}")
        return None

# ----------------------------------------------------------------------
# 3. Diagnostic VBD reference + sub-breakdown dark-current metrics
#
# Main metrics use the n5 ionization-integral VBD.
# The old inverse-fit remains only as a diagnostic output.
# ----------------------------------------------------------------------

vbd_diag = load_diagnostic_vbd()

vbd_fit, primary_exponent, primary_fit_max, i_primary = vbd_selfcontained(
    v,
    ia
)

if i_primary is None:
    i_primary = np.full_like(v, np.nan)

inverse_fit_points = inverse_fit_point_count(
    ia,
    i_primary
)

vbd_i1e9 = vbd_rule1(v, ia, 1.0e-9)
vbd_i1e6 = vbd_rule1(v, ia, 1.0e-6)

if np.isfinite(vbd_diag):
    vbd_ref = float(vbd_diag)
    vbd_ref_source = "diagnostic-ionization-integral"
elif np.isfinite(vbd_fit):
    vbd_ref = float(vbd_fit)
    vbd_ref_source = "legacy-inverse-fit-fallback"
else:
    vbd_ref = np.nan
    vbd_ref_source = "unavailable"

multiplication = np.full_like(v, np.nan)
valid_m = np.isfinite(i_primary) & (i_primary > 0.0) & (ia > 0.0)
multiplication[valid_m] = ia[valid_m] / i_primary[valid_m]

fit_ok = float(
    np.isfinite(vbd_fit) and
    np.isfinite(primary_exponent) and
    inverse_fit_points >= 3 and
    1.0 <= primary_exponent <= 1.6
)

diag_vbd_ok = float(np.isfinite(vbd_diag))

v50 = 0.50 * vbd_ref if np.isfinite(vbd_ref) else np.nan
v90 = 0.90 * vbd_ref if np.isfinite(vbd_ref) else np.nan

i50 = interp_inside(v, ia, v50) if np.isfinite(v50) else np.nan
i90 = interp_inside(v, ia, v90) if np.isfinite(v90) else np.nan
i156 = interp_inside(v, ia, 15.6)

m90 = (
    interp_inside(v, multiplication, v90)
    if np.isfinite(v90) and fit_ok > 0.5
    else np.nan
)

i50_density = i50 / ACTIVE_AREA_UM2 if np.isfinite(i50) else np.nan
i90_density = i90 / ACTIVE_AREA_UM2 if np.isfinite(i90) else np.nan
i156_density = i156 / ACTIVE_AREA_UM2 if np.isfinite(i156) else np.nan

srh90 = interp_inside(v, srh_eq_a, v90) if np.isfinite(v90) else np.nan
btbt90 = interp_inside(v, btbt_eq_a, v90) if np.isfinite(v90) else np.nan
aval90 = interp_inside(v, aval_eq_a, v90) if np.isfinite(v90) else np.nan

valid_components90 = [
    value for value in (srh90, btbt90, aval90)
    if np.isfinite(value)
]
component_sum90 = (
    float(np.sum(valid_components90))
    if valid_components90
    else np.nan
)

srh90_pct = (
    100.0 * srh90 / component_sum90
    if np.isfinite(srh90) and np.isfinite(component_sum90) and component_sum90 > 0.0
    else np.nan
)
btbt90_pct = (
    100.0 * btbt90 / component_sum90
    if np.isfinite(btbt90) and np.isfinite(component_sum90) and component_sum90 > 0.0
    else np.nan
)
aval90_pct = (
    100.0 * aval90 / component_sum90
    if np.isfinite(aval90) and np.isfinite(component_sum90) and component_sum90 > 0.0
    else np.nan
)

kcl_abs = np.abs(ia_signed + ic_signed + is_signed)
kcl_scale = np.maximum.reduce(
    [
        np.abs(ia_signed),
        np.abs(ic_signed),
        np.abs(is_signed),
        np.full_like(v, 1.0e-30)
    ]
)
kcl_rel_max = float(np.max(kcl_abs / kcl_scale))

dark_ok = float(
    np.isfinite(vbd_ref) and
    np.isfinite(i50) and
    np.isfinite(i90) and
    np.isfinite(i156) and
    np.isfinite(v50) and
    np.isfinite(v90) and
    np.max(v) >= v90 and
    np.max(v) >= 15.6
)
